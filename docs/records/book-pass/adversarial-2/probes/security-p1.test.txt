safe("AH1 ahead evaluation + attempt performs effect?", () => {
  const { k, call, kp, write, wrote, steps, add } = setup();
  const c = k.cell();
  const r = k.evaluate(call(kp("then"), call(add, c, byte(k, 1)), call(kp("attempt"), call(write, byte(k, 6)), steps(100))), k.scope());
  log("ahead attempt: wrote", wrote(), k.kind(r.value));
  const f = k.composedFunction([], call(write, byte(k, 5)));
  k.evaluate(call(kp("then"), call(add, c, byte(k, 1)), call(f)), k.scope());
  log("ahead call: wrote", wrote());
  k.evaluate(call(kp("then"), call(add, c, byte(k, 1)), call(kp("get"), k.structure([[null, call(write, byte(k, 4))]]), k.bits(new Uint8Array(8)))), k.scope());
  log("ahead get: wrote", wrote());
  k.evaluate(call(kp("then"), call(add, c, byte(k, 1)), call(kp("block"), k.structure([[k.symbol("m.w"), call(write, byte(k, 3))]]), k.symbol("m.w"))), k.scope());
  log("ahead block: wrote", wrote());
  k.evaluate(call(kp("then"), call(add, c, byte(k, 1)), call(kp("if"), truth(k, true), call(write, byte(k, 2)), byte(k, 0))), k.scope());
  log("ahead if: wrote", wrote());
  k.fill(c, byte(k, 1)); k.advance();
  log("after fill: wrote", wrote());
});
