safe("PC build: free names, stray params, primitive by hash", () => {
  const s = setup(); const { k, call, kp, write, wrote, show, form } = s;
  const sym = (n: string) => k.symbolData ? k.symbolData("description." + n) : k.symbol("description." + n);
  const D = (n: string, ...parts: any[]) => k.structure([[null, sym(n)], ...parts.map((p) => [null, p] as const)]);
  const arr = (...xs: any[]) => k.structure(xs.map((x) => [null, x] as const));
  const w64 = (n: number) => { const b = new Uint8Array(8); b[7] = n; return k.bits(b); };
  // learn real shapes from a real description
  const p = k.param("p");
  const sample = k.composedFunction([p], call(s.add, p, k.symbol("m.secret")));
  const real = k.evaluate(call(kp("describe"), k.inspectionRight(), sample), k.scope()).value;
  log("sample description:", show(real));
  const ents = k.entriesOf(real);
  // 1. function with a free name `m.secret`, built by attacker code, applied in victim block scope
  const desc = D("function", k.entriesOf(real)[1][1], D("name", k.symbolData("m.secret")));
  const attackerScope = k.scope();
  const built = k.evaluate(call(kp("build"), desc), attackerScope);
  log("build fn with free name: refuted", built.refuted, k.reasonOf(built.value), k.kind(built.value));
  const evil = k.composedFunction([], call(kp("build"), desc));  // attacker function returning a built function
  k.bind(attackerScope, k.symbol("att.evil"), evil);
  const evilV = k.evaluate(k.symbol("att.evil"), attackerScope).value;
  // victim: block({m.secret: 77}, evil()(0))
  const r = k.evaluate(call(kp("block"), k.structure([[k.symbol("m.secret"), byte(k, 77)]]), call(call(evilV), byte(k, 0))), k.scope());
  log("victim calls evil()(0) inside a block binding m.secret=77 ->", k.kind(r.value), k.kind(r.value) === "bits" ? num(k, r.value) : "", "refuted", r.refuted, k.reasonOf(r.value));
  // 2. stray param
  const stray = k.evaluate(call(kp("build"), D("function", k.entriesOf(real)[1][1], D("param", w64(1), w64(0)))), k.scope());
  log("build with a param one function out: refuted", stray.refuted, k.reasonOf(stray.value));
  // 3. primitive by hash, never handed
  const k9 = k.primitiveFunction({ purity: "pure", hash: hash(99), name: "clock", arity: 0, run: (_a, c) => c.bits(Uint8Array.of(Date.now() & 255)) });
  void k9;
  const byHash = k.evaluate(call(kp("build"), D("function", w64(0), D("apply", D("primitive", k.bits(new TextEncoder().encode(hash(99)))), arr()))), k.scope());
  log("build names a pure primitive by hash it was never handed: refuted", byHash.refuted, k.reasonOf(byHash.value), k.kind(byHash.value));
  if (!byHash.refuted) { const v = k.evaluate(call(byHash.value), k.scope()); log("  and calls it:", k.kind(v.value), v.refuted); }
  // 4. key form
  const keyForm = k.evaluate(call(kp("build"), D("function", w64(0), D("key", k.bits(new TextEncoder().encode(hash(7))), w64(0)))), k.scope());
  log("build with a key form: refuted", keyForm.refuted, k.reasonOf(keyForm.value));
  void write; void wrote; void form; void ents;
});

safe("PD thunk-wrap an argument and describe it with the wrapper's own key", () => {
  const s = setup(); const { k, call, kp, add, show } = s;
  const [key, a, f] = [k.param("key"), k.param("a"), k.param("f")];
  const A = k.module(k.composedFunction([key], k.structure([[null, k.composedFunction([a], k.composedFunction([], a))], [null, k.composedFunction([f], call(kp("describe"), key, f))]])), 1);
  const inst = k.entriesOf(k.evaluate(call(A), k.scope()).value).map(([, v]) => k.evaluate(v, k.scope()).value);
  const [wrap, descMine] = inst;
  const [kv, w, pw] = [k.param("kv"), k.param("wrap"), k.param("pw")];
  // victim module: (kv, wrap) => (pw => wrap(add(add(pw, 42), 1)))  -- a private computation over a secret constant 42
  const helper = k.composedFunction([pw], call(add, pw, byte(k, 42)));
  const V = k.module(k.composedFunction([kv, w], call(w, call(add, call(helper, byte(k, 7)), byte(k, 1)))), 1);
  const thunk = k.evaluate(call(V, wrap), k.scope());
  log("thunk kind", k.kind(thunk.value), "refuted", thunk.refuted);
  const d = k.evaluate(call(descMine, thunk.value), k.scope());
  log("attacker describes its thunk:", d.refuted ? k.reasonOf(d.value) : show(d.value));
});
